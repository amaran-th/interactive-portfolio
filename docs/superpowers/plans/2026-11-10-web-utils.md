```ts
// 색 변환과 CSS 문자열. 색상 스포이트·팔레트 추출기가 같이 쓴다.
// node:test가 그대로 실행하도록 다른 모듈을 import하지 않는다.

export type Rgb = { r: number; g: number; b: number };

const hex2 = (n: number) => n.toString(16).padStart(2, "0");

export function rgbToHex({ r, g, b }: Rgb): string {
  return `#${hex2(r)}${hex2(g)}${hex2(b)}`;
}

export function hexToRgb(hex: string): Rgb | null {
  const m = /^#?([0-9a-f]{3}|[0-9a-f]{6})$/i.exec(hex.trim());
  if (!m) return null;
  let h = m[1];
  if (h.length === 3) h = h.split("").map((c) => c + c).join("");
  const n = parseInt(h, 16);
  return { r: (n >> 16) & 255, g: (n >> 8) & 255, b: n & 255 };
}

// h는 0~359, s·l은 0~100 정수로 반올림한다.
export function rgbToHsl({ r, g, b }: Rgb): { h: number; s: number; l: number } {
  const rn = r / 255;
  const gn = g / 255;
  const bn = b / 255;
  const max = Math.max(rn, gn, bn);
  const min = Math.min(rn, gn, bn);
  const l = (max + min) / 2;
  const d = max - min;
  if (d === 0) return { h: 0, s: 0, l: Math.round(l * 100) };
  const s = d / (1 - Math.abs(2 * l - 1));
  let h: number;
  if (max === rn) h = ((gn - bn) / d) % 6;
  else if (max === gn) h = (bn - rn) / d + 2;
  else h = (rn - gn) / d + 4;
  h *= 60;
  if (h < 0) h += 360;
  return { h: Math.round(h) % 360, s: Math.round(s * 100), l: Math.round(l * 100) };
}

export const formatHex = rgbToHex;

export function formatRgb({ r, g, b }: Rgb): string {
  return `rgb(${r} ${g} ${b})`;
}

export function formatHsl(rgb: Rgb): string {
  const { h, s, l } = rgbToHsl(rgb);
  return `hsl(${h} ${s}% ${l}%)`;
}

// 견본 위 글자를 흰색으로 쓸지 고를 때 쓴다(Rec. 601 luma).
export function isDark({ r, g, b }: Rgb): boolean {
  return 0.299 * r + 0.587 * g + 0.114 * b < 140;
}

// EyeDropper 결과처럼 HEX나 rgb() 문자열로 들어온 색을 읽는다.
export function parseColor(text: string): Rgb | null {
  const fromHex = hexToRgb(text);
  if (fromHex) return fromHex;
  const m = /^rgba?\(\s*(\d{1,3})[\s,]+(\d{1,3})[\s,]+(\d{1,3})/i.exec(text.trim());
  if (!m) return null;
  const [r, g, b] = [m[1], m[2], m[3]].map(Number);
  if (r > 255 || g > 255 || b > 255) return null;
  return { r, g, b };
}
```

```ts
export type PaletteEntry = { hex: string; ratio: number };

// RGBA 픽셀 배열(ImageData.data 등)에서 대표색을 뽑아 많이 쓰인 순으로
// 돌려준다. 알파 128 미만은 투명으로 보고 뺀다. ratio는 남은 픽셀 대비 비율.
export function extractPalette(
  data: ArrayLike<number>,
  maxColors: number,
): PaletteEntry[] {
  const palette: string[] = [];
  const index = new Map<string, number>();
  const pixels: number[] = [];
  const h2 = (n: number) => n.toString(16).padStart(2, "0");
  for (let o = 0; o + 3 < data.length; o += 4) {
    if (data[o + 3] < 128) continue;
    const hex = `#${h2(data[o])}${h2(data[o + 1])}${h2(data[o + 2])}`;
    let i = index.get(hex);
    if (i === undefined) {
      i = palette.length;
      palette.push(hex);
      index.set(hex, i);
    }
    pixels.push(i);
  }
  if (pixels.length === 0) return [];
  const q = quantizeMedianCut(palette, pixels, maxColors);
  const counts = new Array<number>(q.palette.length).fill(0);
  for (const p of q.pixels) counts[p]++;
  return q.palette
    .map((hex, i) => ({ hex, ratio: counts[i] / pixels.length }))
    .sort((a, b) => b.ratio - a.ratio);
}
```

```ts
// 문자 인코딩 계산. 화면(TextEncoding.tsx)과 node:test가 같이 쓴다.
// node:test가 그대로 실행하도록 다른 모듈을 import하지 않는다.

export type Encoding = "utf-8" | "utf-16be" | "euc-kr" | "ascii";
export type Radix = 16 | 10 | 2;
export type EucKrTable = Map<number, [number, number]>;
export type ByteDecoder = { decode(input: Uint8Array): string };
export type ParseResult = { values: number[]; error: string | null };
export type DecodeResult = { text: string; error: string | null };

const hex = (n: number, width = 2) =>
  n.toString(16).toUpperCase().padStart(width, "0");

const isHighSurrogate = (u: number) => u >= 0xd800 && u <= 0xdbff;
const isLowSurrogate = (u: number) => u >= 0xdc00 && u <= 0xdfff;
const isSurrogate = (u: number) => u >= 0xd800 && u <= 0xdfff;

export function codePoints(text: string): number[] {
  return Array.from(text, (c) => c.codePointAt(0)!);
}

export function utf8Bytes(cp: number): number[] {
  if (cp < 0x80) return [cp];
  if (cp < 0x800) return [0xc0 | (cp >> 6), 0x80 | (cp & 0x3f)];
  if (cp < 0x10000)
    return [0xe0 | (cp >> 12), 0x80 | ((cp >> 6) & 0x3f), 0x80 | (cp & 0x3f)];
  return [
    0xf0 | (cp >> 18),
    0x80 | ((cp >> 12) & 0x3f),
    0x80 | ((cp >> 6) & 0x3f),
    0x80 | (cp & 0x3f),
  ];
}

export function utf16Units(cp: number): number[] {
  if (cp < 0x10000) return [cp];
  const v = cp - 0x10000;
  return [0xd800 | (v >> 10), 0xdc00 | (v & 0x3ff)];
}

// 빅 엔디언
export function utf16Bytes(cp: number): number[] {
  return utf16Units(cp).flatMap((u) => [u >> 8, u & 0xff]);
}

export function asciiBytes(cp: number): number[] | null {
  return cp < 0x80 ? [cp] : null;
}

// 브라우저는 EUC-KR 인코딩을 지원하지 않아서, 디코더로 KS X 1001 영역
// (A1~FE × A1~FE)을 한 번 훑어 역방향 표를 만든다. 브라우저의 euc-kr
// 디코더는 실제로 CP949라 확장 영역도 풀지만, 이 표에는 넣지 않는다.
export function buildEucKrTable(decoder: ByteDecoder): EucKrTable {
  const table: EucKrTable = new Map();
  const buf = new Uint8Array(2);
  for (let b1 = 0xa1; b1 <= 0xfe; b1++) {
    for (let b2 = 0xa1; b2 <= 0xfe; b2++) {
      buf[0] = b1;
      buf[1] = b2;
      let s: string;
      try {
        s = decoder.decode(buf);
      } catch {
        continue;
      }
      const cps = codePoints(s);
      if (cps.length !== 1 || cps[0] === 0xfffd) continue;
      if (!table.has(cps[0])) table.set(cps[0], [b1, b2]);
    }
  }
  return table;
}

export function eucKrBytes(cp: number, table: EucKrTable): number[] | null {
  if (cp < 0x80) return [cp];
  const pair = table.get(cp);
  return pair ? [...pair] : null;
}

export function formatCodePoint(cp: number): string {
  return `U+${hex(cp, 4)}`;
}

// 표의 글자 칸에 보여 줄 모양. 보이지 않는 문자는 기호나 코드로 바꾼다.
export function displayChar(cp: number): string {
  if (cp === 0x20) return "␣";
  if (cp === 0x0a) return "↵";
  if (cp < 0x20 || (cp >= 0x7f && cp < 0xa0)) return formatCodePoint(cp);
  return String.fromCodePoint(cp);
}

function formatNumber(n: number, radix: Radix, bits: 8 | 16): string {
  if (radix === 10) return String(n);
  if (radix === 2) return n.toString(2).padStart(bits, "0");
  return hex(n, bits / 4);
}

export function formatBytes(bytes: number[], radix: Radix): string {
  return bytes.map((b) => formatNumber(b, radix, 8)).join(" ");
}

export function formatUnits16(units: number[], radix: Radix): string {
  return units.map((u) => formatNumber(u, radix, 16)).join(" ");
}

const DIGITS: Record<Radix, RegExp> = {
  16: /^[0-9a-f]+$/i,
  10: /^[0-9]+$/,
  2: /^[01]+$/,
};
const RADIX_NAME: Record<Radix, string> = { 16: "16진수", 10: "10진수", 2: "2진수" };

// 바이트열 입력을 읽는다. 16진수는 0x·\x 접두사를 지우고, 붙여 쓴 값
// (EAB080)은 2자리씩, 2진수는 8자리씩 자른다. 구분자는 공백·쉼표.
export function parseBytes(input: string, radix: Radix = 16): ParseResult {
  let s = input;
  if (radix === 16) s = s.replace(/\\x|0x/gi, " ");
  if (radix === 2) s = s.replace(/0b/gi, " ");
  const tokens = s.split(/[\s,]+/).filter(Boolean);
  const width = radix === 16 ? 2 : radix === 2 ? 8 : 0;
  const values: number[] = [];
  for (let i = 0; i < tokens.length; i++) {
    const tok = tokens[i];
    const where = `${i + 1}번째 값 ${tok}`;
    if (!DIGITS[radix].test(tok))
      return { values, error: `${where}: ${RADIX_NAME[radix]}가 아니에요` };
    const chunks: string[] = [];
    if (width && tok.length > width) {
      if (tok.length % width !== 0)
        return { values, error: `${where}: 자리 수가 ${width}의 배수가 아니에요` };
      for (let k = 0; k < tok.length; k += width) chunks.push(tok.slice(k, k + width));
    } else {
      chunks.push(tok);
    }
    for (const c of chunks) {
      const v = parseInt(c, radix);
      if (v > 255) return { values, error: `${where}: 한 바이트(0~255)를 넘어요` };
      values.push(v);
    }
  }
  return { values, error: null };
}

export function looksLikeCodePoints(input: string): boolean {
  return /U\+|\\u|&#/i.test(input);
}

const CP_TOKEN =
  /U\+([0-9a-f]{1,6})|\\u\{([0-9a-f]{1,6})\}|\\u([0-9a-f]{4})|&#x([0-9a-f]{1,6});?|&#([0-9]{1,7});?/gi;

// 😀처럼 나뉜 서로게이트 쌍을 코드포인트 하나로 합친다.
function combineSurrogates(units: number[]): ParseResult {
  const values: number[] = [];
  for (let i = 0; i < units.length; i++) {
    const u = units[i];
    if (isHighSurrogate(u) && i + 1 < units.length && isLowSurrogate(units[i + 1])) {
      values.push(0x10000 + ((u - 0xd800) << 10) + (units[i + 1] - 0xdc00));
      i++;
      continue;
    }
    if (isSurrogate(u))
      return { values, error: `${i + 1}번째 코드 ${formatCodePoint(u)}: 짝이 없는 서로게이트예요` };
    values.push(u);
  }
  return { values, error: null };
}

export function parseCodePoints(input: string): ParseResult {
  const units: number[] = [];
  let last = 0;
  const fail = (error: string): ParseResult => ({
    values: combineSurrogates(units).values,
    error,
  });
  for (const m of input.matchAll(CP_TOKEN)) {
    const gap = input.slice(last, m.index).replace(/[\s,]+/g, "");
    if (gap) return fail(`${gap}: 코드로 읽지 못했어요`);
    const v =
      m[5] !== undefined ? parseInt(m[5], 10) : parseInt(m[1] ?? m[2] ?? m[3] ?? m[4], 16);
    if (v > 0x10ffff)
      return fail(`${units.length + 1}번째 코드 ${m[0]}: 유니코드 범위(U+10FFFF)를 넘어요`);
    units.push(v);
    last = m.index + m[0].length;
  }
  const tail = input.slice(last).replace(/[\s,]+/g, "");
  if (tail) return fail(`${tail}: 코드로 읽지 못했어요`);
  return combineSurrogates(units);
}

function decodeAscii(bytes: number[]): DecodeResult {
  let text = "";
  for (let i = 0; i < bytes.length; i++) {
    if (bytes[i] > 0x7f)
      return { text, error: `${i + 1}번째 바이트 ${hex(bytes[i])}: ASCII 범위(00~7F)를 넘어요` };
    text += String.fromCharCode(bytes[i]);
  }
  return { text, error: null };
}

const UTF8_MIN = [0, 0x80, 0x800, 0x10000];

function decodeUtf8(bytes: number[]): DecodeResult {
  let text = "";
  let i = 0;
  while (i < bytes.length) {
    const b = bytes[i];
    if (b < 0x80) {
      text += String.fromCharCode(b);
      i++;
      continue;
    }
    let need: number;
    let cp: number;
    if (b >= 0xc0 && b <= 0xdf) {
      need = 1;
      cp = b & 0x1f;
    } else if (b >= 0xe0 && b <= 0xef) {
      need = 2;
      cp = b & 0x0f;
    } else if (b >= 0xf0 && b <= 0xf7) {
      need = 3;
      cp = b & 0x07;
    } else {
      return { text, error: `${i + 1}번째 바이트 ${hex(b)}: UTF-8 글자의 첫 바이트가 될 수 없어요` };
    }
    if (i + need >= bytes.length)
      return { text, error: `${i + 1}번째 바이트 ${hex(b)}: 뒤에 바이트 ${need}개가 더 필요해요` };
    for (let k = 1; k <= need; k++) {
      const c = bytes[i + k];
      if ((c & 0xc0) !== 0x80)
        return { text, error: `${i + k + 1}번째 바이트 ${hex(c)}: 이어지는 바이트(80~BF)여야 해요` };
      cp = (cp << 6) | (c & 0x3f);
    }
    if (cp < UTF8_MIN[need] || cp > 0x10ffff || isSurrogate(cp))
      return { text, error: `${i + 1}번째 바이트부터: 올바르지 않은 UTF-8 글자예요` };
    text += String.fromCodePoint(cp);
    i += need + 1;
  }
  return { text, error: null };
}

function decodeUtf16(bytes: number[]): DecodeResult {
  let text = "";
  const units = Math.floor(bytes.length / 2);
  const unitAt = (k: number) => (bytes[2 * k] << 8) | bytes[2 * k + 1];
  for (let k = 0; k < units; k++) {
    const u = unitAt(k);
    if (isHighSurrogate(u) && k + 1 < units && isLowSurrogate(unitAt(k + 1))) {
      text += String.fromCodePoint(0x10000 + ((u - 0xd800) << 10) + (unitAt(k + 1) - 0xdc00));
      k++;
      continue;
    }
    if (isSurrogate(u))
      return { text, error: `${2 * k + 1}번째 바이트부터: 짝이 없는 서로게이트 ${hex(u, 4)}예요` };
    text += String.fromCharCode(u);
  }
  if (bytes.length % 2)
    return { text, error: "바이트 수가 홀수예요. UTF-16은 2바이트씩 읽어요" };
  return { text, error: null };
}

function decodeEucKr(bytes: number[], decoder: ByteDecoder | null): DecodeResult {
  if (!decoder) return { text: "", error: "이 브라우저는 EUC-KR 해독을 지원하지 않아요" };
  let text = "";
  let i = 0;
  while (i < bytes.length) {
    const b = bytes[i];
    if (b < 0x80) {
      text += String.fromCharCode(b);
      i++;
      continue;
    }
    if (i + 1 >= bytes.length)
      return { text, error: `${i + 1}번째 바이트 ${hex(b)}: 뒤에 바이트 1개가 더 필요해요` };
    const pairError = {
      text,
      error: `${i + 1}번째 바이트부터: EUC-KR에 없는 조합(${hex(b)} ${hex(bytes[i + 1])})이에요`,
    };
    let ch: string;
    try {
      ch = decoder.decode(new Uint8Array([b, bytes[i + 1]]));
    } catch {
      return pairError;
    }
    if (ch.includes("�")) return pairError;
    text += ch;
    i += 2;
  }
  return { text, error: null };
}

export function decodeBytes(
  bytes: number[],
  encoding: Encoding,
  eucKr: ByteDecoder | null = null,
): DecodeResult {
  switch (encoding) {
    case "ascii":
      return decodeAscii(bytes);
    case "utf-8":
      return decodeUtf8(bytes);
    case "utf-16be":
      return decodeUtf16(bytes);
    case "euc-kr":
      return decodeEucKr(bytes, eucKr);
  }
}
```

```ts
"use client";

import { useCallback, useEffect, useRef, useState } from "react";

const SHOW_MS = 1500;

// 클립보드에 복사하고 "복사됨"·"복사하지 못했어요"를 잠깐 보여 준다.
export function useCopyFeedback() {
  const [message, setMessage] = useState<string | null>(null);
  const timer = useRef<ReturnType<typeof setTimeout> | null>(null);

  useEffect(
    () => () => {
      if (timer.current) clearTimeout(timer.current);
    },
    [],
  );

  const copy = useCallback(async (text: string) => {
    let ok = false;
    try {
      if (navigator.clipboard) {
        await navigator.clipboard.writeText(text);
        ok = true;
      }
    } catch {
      ok = false;
    }
    setMessage(ok ? "복사됨" : "복사하지 못했어요");
    if (timer.current) clearTimeout(timer.current);
    timer.current = setTimeout(() => setMessage(null), SHOW_MS);
  }, []);

  return { copy, message };
}
```

```tsx
function EncodeView({ radix, eucKr, onCopy }: ViewProps) {

function DecodeView({ radix, eucKr, onCopy }: ViewProps) {
  const [input, setInput] = useState("");
  const [encoding, setEncoding] = useState<Encoding>("utf-8");
  const asCodePoints = looksLikeCodePoints(input);
  const result = useMemo(() => {
    if (!input.trim()) return null;
    if (asCodePoints) {
      const p = parseCodePoints(input);
      return { text: p.values.map((v) => String.fromCodePoint(v)).join(""), error: p.error };
    }
    const p = parseBytes(input, radix);
    const d = decodeBytes(p.values, encoding, eucKr?.decoder ?? null);
    return { text: d.text, error: p.error ?? d.error };
  }, [input, asCodePoints, radix, encoding, eucKr]);

  return (
    <>
      <textarea
        aria-label="바이트 또는 코드"
        value={input}
        onChange={(e) => setInput(e.target.value)}
        placeholder="EA B0 80 또는 U+AC00"
        rows={3}
        className={`w-full resize-y p-3 font-mono text-base focus:outline-none focus-visible:ring-2 focus-visible:ring-main ${STICKER}`}
      />
      <fieldset className="flex flex-wrap items-center gap-2 text-sm" disabled={asCodePoints}>
        <legend className="sr-only">인코딩</legend>
        {ENCODINGS.map((e) => (
          <label
            key={e.id}
            className={`cursor-pointer rounded-full border-2 border-point px-3 py-1 font-bold ${
              encoding === e.id ? "bg-point text-white" : "bg-white text-point"
            } ${asCodePoints ? "opacity-40" : ""}`}
          >
            <input
              type="radio"
              name="encoding"
              value={e.id}
              checked={encoding === e.id}
              onChange={() => setEncoding(e.id)}
              className="sr-only"
            />
            {e.label}
          </label>
        ))}
        {asCodePoints && (
          <span className="text-xs text-ink/60">코드포인트 형식이라 인코딩 없이 읽어요</span>
        )}
      </fieldset>
      {result && (
        <div className="rounded-lg border-2 border-point bg-white p-4">
          <p
            aria-label="복원한 글자"
            className="min-h-8 text-2xl break-all whitespace-pre-wrap text-ink"
          >
            {result.text}
          </p>
          {result.error && (
            <p role="alert" className="mt-2 text-sm font-bold text-red-600">
              {result.error}
            </p>
          )}

            <button
              type="button"
              onClick={() => onCopy(result.text)}

            >
              글자 복사
            </button>
          )}
        </div>
      )}
    </>
  );
}
```

```ts
"use client";

import { useCallback, useEffect, useRef, useState } from "react";
import type React from "react";

export type ImageInput = {
  image: HTMLImageElement | null;
  error: string | null;
  dragging: boolean;
  load: (file: File | null | undefined) => void;
  dropProps: {
    onDragOver: (e: React.DragEvent) => void;
    onDragLeave: () => void;
    onDrop: (e: React.DragEvent) => void;
  };
};

// 파일 선택·끌어다 놓기·붙여넣기(Ctrl/⌘+V)로 이미지 한 장을 받는다.
// 붙여넣기는 페이지 전체에서 받되, 글자 입력칸에 붙여 넣는 건 건드리지 않는다.
export function useImageInput(): ImageInput {
  const [image, setImage] = useState<HTMLImageElement | null>(null);
  const [error, setError] = useState<string | null>(null);
  const [dragging, setDragging] = useState(false);
  const urlRef = useRef<string | null>(null);

  const load = useCallback((file: File | null | undefined) => {
    if (!file) return;
    if (!file.type.startsWith("image/")) {
      setError("이미지 파일만 넣을 수 있어요");
      return;
    }
    const url = URL.createObjectURL(file);
    const img = new Image();
    img.onload = () => {
      if (urlRef.current) URL.revokeObjectURL(urlRef.current);
      urlRef.current = url;
      setImage(img);
      setError(null);
    };
    img.onerror = () => {
      URL.revokeObjectURL(url);
      setError("이미지를 읽지 못했어요");
    };
    img.src = url;
  }, []);

  useEffect(() => {
    const onPaste = (e: ClipboardEvent) => {
      const t = e.target as HTMLElement | null;
      if (t && (t.tagName === "INPUT" || t.tagName === "TEXTAREA" || t.isContentEditable)) return;
      const file = Array.from(e.clipboardData?.files ?? []).find((f) =>
        f.type.startsWith("image/"),
      );
      if (file) {
        e.preventDefault();
        load(file);
      } else {
        setError("붙여 넣은 내용에 이미지가 없어요");
      }
    };
    window.addEventListener("paste", onPaste);
    return () => window.removeEventListener("paste", onPaste);
  }, [load]);

  useEffect(
    () => () => {
      if (urlRef.current) URL.revokeObjectURL(urlRef.current);
    },
    [],
  );

  return {
    image,
    error,
    dragging,
    load,
    dropProps: {
      onDragOver: (e) => {
        e.preventDefault();
        setDragging(true);
      },
      onDragLeave: () => setDragging(false),
      onDrop: (e) => {
        e.preventDefault();
        setDragging(false);
        load(e.dataTransfer.files[0]);
      },
    },
  };
}
```

```tsx
"use client";

import { ImagePlus } from "lucide-react";
import { useRef } from "react";
import type { ImageInput } from "./useImageInput";

// 이미지를 넣는 영역. compact면 이미지가 이미 있을 때 쓰는 작은 '다른 이미지' 줄.
export default function ImageDropZone({
  input,
  compact = false,
}: {
  input: ImageInput;
  compact?: boolean;
}) {
  const fileRef = useRef<HTMLInputElement>(null);
  return (
    <div
      {...input.dropProps}
      className={`flex flex-col items-center justify-center gap-2 rounded-lg border-2 border-dashed p-4 text-center text-sm transition-colors ${
        input.dragging ? "border-main bg-white" : "border-point/30 bg-white/60"
      } ${compact ? "" : "min-h-56 flex-1"}`}
    >
      {!compact && <ImagePlus aria-hidden className="size-8 text-point/60" />}
      <button
        type="button"
        onClick={() => fileRef.current?.click()}
        className="rounded-md border-2 border-point bg-white px-3 py-1.5 font-bold text-point shadow-[2px_2px_0_0_var(--color-point)] hover:bg-shade"
      >
        {compact ? "다른 이미지" : "이미지 고르기"}
      </button>
      <p className="text-xs text-ink/60">끌어다 놓거나 Ctrl/⌘+V로 붙여 넣어도 돼요</p>
      {input.error && (
        <p role="alert" className="text-xs font-bold text-red-600">
          {input.error}
        </p>
      )}
      <input
        ref={fileRef}
        type="file"
        accept="image/*"
        className="hidden"
        onChange={(e) => {
          input.load(e.target.files?.[0]);
          e.target.value = "";
        }}
      />
    </div>
  );
}
```

```tsx
"use client";

import { Pipette } from "lucide-react";
import { useCallback, useEffect, useRef, useState, useSyncExternalStore } from "react";
import type React from "react";
import { type Rgb, formatHex, formatHsl, formatRgb, isDark, parseColor } from "../_shared/color";
import CopyToast from "../_shared/CopyToast";
import ImageDropZone from "../_shared/ImageDropZone";
import { useCopyFeedback } from "../_shared/useCopyFeedback";
import { useImageInput } from "../_shared/useImageInput";

const MAX_RECENT = 10;
const LOUPE_CELLS = 11;
const LOUPE_CELL_PX = 10;
const LOUPE_PX = LOUPE_CELLS * LOUPE_CELL_PX;
const LOUPE_OFFSET = 16;

type EyeDropperCtor = new () => { open: () => Promise<{ sRGBHex: string }> };
type Hover = { x: number; y: number; clientX: number; clientY: number; hex: string };

const noopSubscribe = () => () => {};

const FORMATS = [
  { label: "HEX", format: formatHex },
  { label: "RGB", format: formatRgb },
  { label: "HSL", format: formatHsl },
];

export default function ColorPicker() {
  const input = useImageInput();
  const { copy, message } = useCopyFeedback();
  const canvasRef = useRef<HTMLCanvasElement>(null);
  const loupeRef = useRef<HTMLCanvasElement>(null);
  const [current, setCurrent] = useState<Rgb | null>(null);
  const [recent, setRecent] = useState<Rgb[]>([]);
  const [notice, setNotice] = useState<string | null>(null);
  const [hover, setHover] = useState<Hover | null>(null);
  // 서버에서는 false. EyeDropper API는 데스크톱 크로미움 브라우저에만 있다.
  const screenPickSupported = useSyncExternalStore(
    noopSubscribe,
    () => "EyeDropper" in window,
    () => false,
  );

  // 원본 해상도 그대로 캔버스에 그려 두고 거기서 픽셀을 읽는다.
  useEffect(() => {
    const canvas = canvasRef.current;
    const img = input.image;
    if (!canvas || !img) return;
    canvas.width = img.naturalWidth;
    canvas.height = img.naturalHeight;
    canvas.getContext("2d", { willReadFrequently: true })!.drawImage(img, 0, 0);
  }, [input.image]);

  // 돋보기: 포인터 주변 11×11 픽셀을 칸마다 10px로 키워 그린다.
  useEffect(() => {
    const loupe = loupeRef.current;
    const canvas = canvasRef.current;
    if (!hover || !loupe || !canvas) return;
    const half = (LOUPE_CELLS - 1) / 2;
    const data = canvas
      .getContext("2d")!
      .getImageData(hover.x - half, hover.y - half, LOUPE_CELLS, LOUPE_CELLS);
    const tmp = document.createElement("canvas");
    tmp.width = LOUPE_CELLS;
    tmp.height = LOUPE_CELLS;
    tmp.getContext("2d")!.putImageData(data, 0, 0);
    const ctx = loupe.getContext("2d")!;
    ctx.imageSmoothingEnabled = false;
    ctx.clearRect(0, 0, LOUPE_PX, LOUPE_PX);
    ctx.drawImage(tmp, 0, 0, LOUPE_PX, LOUPE_PX);
    ctx.strokeStyle = "#0c3675";
    ctx.lineWidth = 2;
    ctx.strokeRect(half * LOUPE_CELL_PX, half * LOUPE_CELL_PX, LOUPE_CELL_PX, LOUPE_CELL_PX);
  }, [hover]);

  const choose = useCallback((rgb: Rgb) => {
    setCurrent(rgb);
    setNotice(null);
    setRecent((prev) =>
      [rgb, ...prev.filter((c) => formatHex(c) !== formatHex(rgb))].slice(0, MAX_RECENT),
    );
  }, []);

  function pixelAt(e: React.PointerEvent | React.MouseEvent) {
    const canvas = canvasRef.current!;
    const rect = canvas.getBoundingClientRect();
    const x = Math.floor(((e.clientX - rect.left) / rect.width) * canvas.width);
    const y = Math.floor(((e.clientY - rect.top) / rect.height) * canvas.height);
    const cx = Math.min(canvas.width - 1, Math.max(0, x));
    const cy = Math.min(canvas.height - 1, Math.max(0, y));
    const d = canvas.getContext("2d")!.getImageData(cx, cy, 1, 1).data;
    return { x: cx, y: cy, rgb: { r: d[0], g: d[1], b: d[2] }, alpha: d[3] };
  }

  async function pickFromScreen() {
    const Ctor = (window as unknown as { EyeDropper?: EyeDropperCtor }).EyeDropper;
    if (!Ctor) return;
    try {
      const { sRGBHex } = await new Ctor().open();
      const rgb = parseColor(sRGBHex);
      if (rgb) choose(rgb);
      else setNotice("색을 읽지 못했어요");
    } catch (err) {
      if ((err as DOMException).name !== "AbortError") setNotice("화면에서 색을 뽑지 못했어요");
    }
  }

  const loupeLeft =
    hover && typeof window !== "undefined" && hover.clientX + LOUPE_OFFSET + LOUPE_PX > window.innerWidth
      ? hover.clientX - LOUPE_OFFSET - LOUPE_PX
      : (hover?.clientX ?? 0) + LOUPE_OFFSET;
  const loupeTop =
    hover && typeof window !== "undefined" && hover.clientY + LOUPE_OFFSET + LOUPE_PX + 24 > window.innerHeight
      ? hover.clientY - LOUPE_OFFSET - LOUPE_PX - 24
      : (hover?.clientY ?? 0) + LOUPE_OFFSET;

  return (
    <div className="mx-auto grid min-h-full max-w-6xl gap-5 p-4 sm:p-6 lg:h-full lg:grid-cols-[1fr_20rem]">
      <section aria-label="이미지" className="flex min-h-0 flex-col gap-3">
        {input.image ? (
          <>
            <div className="flex min-h-64 flex-1 items-center justify-center overflow-hidden rounded-lg border-2 border-point bg-white p-2">
              <canvas
                ref={canvasRef}
                aria-label="이미지에서 색 고르기"
                onPointerMove={(e) => {
                  const p = pixelAt(e);
                  setHover({ x: p.x, y: p.y, clientX: e.clientX, clientY: e.clientY, hex: formatHex(p.rgb) });
                }}
                onPointerLeave={() => setHover(null)}
                onClick={(e) => {
                  const p = pixelAt(e);
                  if (p.alpha === 0) setNotice("투명한 곳이라 색이 없어요");
                  else choose(p.rgb);
                }}
                className="max-h-full max-w-full cursor-crosshair touch-none"
                style={{ imageRendering: "pixelated" }}
              />
            </div>
            <ImageDropZone input={input} compact />
          </>
        ) : (
          <ImageDropZone input={input} />
        )}
      </section>

      <aside className="flex flex-col gap-4">
        {screenPickSupported ? (
          <button
            type="button"
            onClick={pickFromScreen}
            className="flex items-center justify-center gap-2 rounded-lg border-2 border-point bg-white px-4 py-2.5 font-bold text-point shadow-[3px_3px_0_0_var(--color-point)] hover:bg-shade"
          >
            <Pipette aria-hidden className="size-4" />
            화면에서 뽑기
          </button>
        ) : (
          <p className="text-xs text-ink/60">
            화면 어디서나 색을 뽑는 기능은 데스크톱 Chrome·Edge에서만 돼요.
          </p>
        )}

        <div
          className="flex h-28 items-center justify-center rounded-lg border-2 border-point text-sm font-bold"
          style={
            current
              ? { backgroundColor: formatHex(current), color: isDark(current) ? "#fff" : "#2e353f" }
              : undefined
          }
        >
          {current ? formatHex(current) : <span className="text-ink/50">이미지를 누르면 여기에 색이 나와요</span>}
        </div>

        {current && (
          <ul className="flex flex-col gap-2">
            {FORMATS.map(({ label, format }) => (
              <li key={label}>
                <button
                  type="button"
                  aria-label={`${label} 복사`}
                  onClick={() => copy(format(current))}
                  className="flex w-full items-center justify-between gap-3 rounded-md border-2 border-point/30 bg-white px-3 py-2 text-left hover:border-point"
                >
                  <span className="text-xs font-bold text-point">{label}</span>
                  <span className="font-mono text-sm">{format(current)}</span>
                </button>
              </li>
            ))}
          </ul>
        )}

        {notice && (
          <p role="status" className="text-sm font-bold text-red-600">
            {notice}
          </p>
        )}

        {recent.length > 0 && (
          <div>
            <h2 className="text-xs font-bold text-point">최근 뽑은 색</h2>
            <ul className="mt-2 flex flex-wrap gap-2">
              {recent.map((c) => (
                <li key={formatHex(c)}>
                  <button
                    type="button"
                    aria-label={`${formatHex(c)} 다시 고르기`}
                    title={formatHex(c)}
                    onClick={() => choose(c)}
                    className="size-8 rounded-md border-2 border-point"
                    style={{ backgroundColor: formatHex(c) }}
                  />
                </li>
              ))}
            </ul>
          </div>
        )}
      </aside>

      {hover && (
        <div
          className="pointer-events-none fixed z-50 flex flex-col items-center gap-1"
          style={{ left: loupeLeft, top: loupeTop }}
        >
          <canvas
            ref={loupeRef}
            width={LOUPE_PX}
            height={LOUPE_PX}
            className="rounded-md border-2 border-point bg-white shadow-lg"
          />
          <span className="rounded bg-point px-1.5 py-0.5 font-mono text-[11px] text-white">
            {hover.hex}
          </span>
        </div>
      )}
      <CopyToast message={message} />
    </div>
  );
}
```

```tsx
"use client";

import { useMemo, useState } from "react";
import { hexToRgb, isDark } from "../_shared/color";
import CopyToast from "../_shared/CopyToast";
import ImageDropZone from "../_shared/ImageDropZone";
import { extractPalette } from "../_shared/quantize";
import { useCopyFeedback } from "../_shared/useCopyFeedback";
import { useImageInput } from "../_shared/useImageInput";

const MIN_COLORS = 3;
const MAX_COLORS = 12;
const DEFAULT_COLORS = 6;
// 계산 전에 긴 변을 이 크기로 줄인다. 큰 사진도 빠르게 계산한다.
const SAMPLE_MAX = 200;

const BUTTON =
  "rounded-md border-2 border-point bg-white px-3 py-1.5 text-sm font-bold text-point shadow-[2px_2px_0_0_var(--color-point)] hover:bg-shade";

export default function PaletteExtractor() {
  const input = useImageInput();
  const { copy, message } = useCopyFeedback();
  const [count, setCount] = useState(DEFAULT_COLORS);

  const pixels = useMemo(() => {
    const img = input.image;
    if (!img) return null;
    const scale = Math.min(1, SAMPLE_MAX / Math.max(img.naturalWidth, img.naturalHeight));
    const w = Math.max(1, Math.round(img.naturalWidth * scale));
    const h = Math.max(1, Math.round(img.naturalHeight * scale));
    const canvas = document.createElement("canvas");
    canvas.width = w;
    canvas.height = h;
    const ctx = canvas.getContext("2d")!;
    ctx.drawImage(img, 0, 0, w, h);
    return ctx.getImageData(0, 0, w, h).data;
  }, [input.image]);

  const palette = useMemo(
    () => (pixels ? extractPalette(pixels, count) : []),
    [pixels, count],
  );

  const cssVars = palette.map((p, i) => `--color-${i + 1}: ${p.hex};`).join("\n");
  const json = JSON.stringify(palette.map((p) => p.hex));

  return (
    <div className="mx-auto grid min-h-full max-w-6xl gap-5 p-4 sm:p-6 lg:h-full lg:grid-cols-[1fr_22rem]">
      <section aria-label="이미지" className="flex min-h-0 flex-col gap-3">
        {input.image ? (
          <>
            <div className="flex min-h-64 flex-1 items-center justify-center overflow-hidden rounded-lg border-2 border-point bg-white p-2">
              {/* eslint-disable-next-line @next/next/no-img-element */}
              <img
                src={input.image.src}
                alt="넣은 이미지"
                className="max-h-full max-w-full object-contain"
              />
            </div>
            <ImageDropZone input={input} compact />
          </>
        ) : (
          <ImageDropZone input={input} />
        )}
      </section>

      <aside className="flex flex-col gap-4">
        <label className="flex flex-col gap-2 text-sm font-bold text-point">
          <span>
            색 개수 <span className="tabular-nums">{count}</span>
          </span>
          <input
            type="range"
            aria-label="색 개수"
            min={MIN_COLORS}
            max={MAX_COLORS}
            value={count}
            onChange={(e) => setCount(Number(e.target.value))}
            className="accent-[var(--color-point)]"
          />
        </label>

        {input.image && palette.length === 0 && (
          <p role="status" className="text-sm text-ink/70">
            투명하지 않은 픽셀이 없어서 색을 찾지 못했어요.
          </p>
        )}

        {palette.length > 0 && (
          <>
            <ul className="grid grid-cols-2 gap-2">
              {palette.map((p) => {
                const rgb = hexToRgb(p.hex)!;
                return (
                  <li key={p.hex}>
                    <button
                      type="button"
                      aria-label={`${p.hex} 복사`}
                      onClick={() => copy(p.hex)}
                      className="flex h-16 w-full flex-col items-start justify-end rounded-md border-2 border-point p-2 text-left transition-transform hover:-rotate-1"
                      style={{ backgroundColor: p.hex, color: isDark(rgb) ? "#fff" : "#2e353f" }}
                    >
                      <span className="font-mono text-sm font-bold">{p.hex}</span>
                      <span className="text-[11px] tabular-nums opacity-80">
                        {Math.round(p.ratio * 1000) / 10}%
                      </span>
                    </button>
                  </li>
                );
              })}
            </ul>
            <div className="flex flex-wrap gap-2">
              <button type="button" onClick={() => copy(cssVars)} className={BUTTON}>
                CSS 변수 복사
              </button>
              <button type="button" onClick={() => copy(json)} className={BUTTON}>
                JSON 복사
              </button>
            </div>
          </>
        )}
      </aside>
      <CopyToast message={message} />
    </div>
  );
}
```

```js
async function makePng(p, w, h, cells) {
  const b64 = await p.evaluate(
    ({ w, h, cells }) => {
      const c = document.createElement("canvas");
      c.width = w;
      c.height = h;
      const ctx = c.getContext("2d");
      for (const [hex, x, y, cw, ch] of cells) {
        ctx.fillStyle = hex;
        ctx.fillRect(x, y, cw, ch);
      }
      return c.toDataURL("image/png").split(",")[1];
    },
    { w, h, cells },
  );
  return { name: "test.png", mimeType: "image/png", buffer: Buffer.from(b64, "base64") };
}
```
