> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 지그재그로 깔린 원형 칩을 뒤집기·빼기·놓기로 바꿔 모자이크를 만드는 3D 창작 도구 `/monorism`을 서비스 앱에 추가한다.

**Architecture:**
- 판 계산은 DOM·three.js 없는 순수 모듈 두 개로 나눠 `node:test`로 검증한다.
  - `layout.ts`: 지그재그 좌표, 점 → 칸, 경로 보간
  - `board.ts`: 상태, 도구, 긋기, 실행 취소, 저장 형식
- 렌더링은 `useChipScene` 훅이 맡는다. three.js `InstancedMesh` 하나로 칩 전체를 그리고, 바뀐 칩만 애니메이션한다.
- `Monorism.tsx`가 포인터·키보드 입력을 판 로직으로 넘기고, 결과를 장면·소리·localStorage로 보낸다.

**Tech Stack:** Next.js 16 App Router, React 19, TypeScript, Tailwind CSS v4, three 0.187, Web Audio API, node:test, Playwright(검증만)

**Spec:** `docs/superpowers/specs/2026-11-10-monorism-design.md`

- 새 의존성은 `three`(dependencies)와 `@types/three`(devDependencies)만 추가한다. 둘 다 `^0.187.0`.
- `three`는 `components/works/10_Monorism/useChipScene.ts`에서만 import한다.

```ts
// 모노리즘 칩 배치 계산. 칩 지름이 1인 좌표계에서 판 중심을 원점으로 둔다.
// x는 오른쪽, y는 줄이 늘어나는 방향(3D에서는 화면 아래쪽, +Z).
// node:test가 그대로 실행하도록 다른 모듈을 import하지 않는다.

export type Point = { x: number; y: number };
export type Cell = { col: number; row: number };

// 홀수 줄을 반 칸 밀면 칩이 벌집처럼 맞물리는 줄 간격
export const ROW_GAP = Math.sqrt(3) / 2;
// 드래그 경로 보간 간격(칩 지름의 1/3) — 빠르게 그어도 칩을 건너뛰지 않는다
export const PATH_STEP = 1 / 3;
// 벌집 배치에서 판 안의 어떤 점도 가장 가까운 칩 중심과 이 거리(1/√3) 안에 있다.
// 이보다 멀면 판 밖으로 본다.
const MAX_DIST_SQ = 1 / 3 + 1e-9;

export function boardSize(w: number, h: number): { width: number; height: number } {
  return { width: w + 0.5, height: (h - 1) * ROW_GAP + 1 };
}

export function cellCenter(col: number, row: number, w: number, h: number): Point {
  const { width, height } = boardSize(w, h);
  return {
    x: col + (row % 2) * 0.5 + 0.5 - width / 2,
    y: row * ROW_GAP + 0.5 - height / 2,
  };
}

// 점에서 가장 가까운 칩 칸. 위아래 두 줄 × 좌우 세 칸만 비교한다.
export function cellAt(p: Point, w: number, h: number): Cell | null {
  const { width, height } = boardSize(w, h);
  const r0 = Math.floor((p.y + height / 2 - 0.5) / ROW_GAP);
  let best: Cell | null = null;
  let bestD = Infinity;
  for (const row of [r0, r0 + 1]) {
    if (row < 0 || row >= h) continue;
    const col0 = Math.round(p.x + width / 2 - 0.5 - (row % 2) * 0.5);
    for (const col of [col0 - 1, col0, col0 + 1]) {
      if (col < 0 || col >= w) continue;
      const c = cellCenter(col, row, w, h);
      const d = (c.x - p.x) ** 2 + (c.y - p.y) ** 2;
      if (d < bestD) {
        bestD = d;
        best = { col, row };
      }
    }
  }
  return bestD <= MAX_DIST_SQ ? best : null;
}

// a에서 b까지 step 간격으로 나눈 점들(a는 이미 처리했으므로 빼고, b는 넣는다).
export function interpolate(a: Point, b: Point, step = PATH_STEP): Point[] {
  const n = Math.max(1, Math.ceil(Math.hypot(b.x - a.x, b.y - a.y) / step));
  const out: Point[] = [];
  for (let i = 1; i <= n; i++) {
    out.push({ x: a.x + ((b.x - a.x) * i) / n, y: a.y + ((b.y - a.y) * i) / n });
  }
  return out;
}
```

```ts
  const h = createHistory();
  const s = beginStroke("flip");

  strokeApply(b, s, 1);
```

```ts
export const EMPTY = 0;
export const FRONT = 1;
export const BACK = 2;

export type Tool = "flip" | "remove" | "place";
export type BulkOp = "flipAll" | "fillAll" | "clearAll";

export type Change = { index: number; before: number; after: number };
export type History = { undo: Change[][]; redo: Change[][] };
export type Stroke = { tool: Tool; touched: Set<number>; changes: Change[] };
export type SavedState = {
  w: number;
  h: number;
  cells: Uint8Array;
  front: string;
  back: string;
  muted: boolean;
};

export function createBoard(w: number, h: number): Board {
  return { w, h, cells: new Uint8Array(w * h).fill(FRONT) };
}

export function applyTool(state: number, tool: Tool): number {
  if (tool === "flip") return state === FRONT ? BACK : state === BACK ? FRONT : EMPTY;
  if (tool === "remove") return EMPTY;
  return state === EMPTY ? FRONT : state;
}

export function beginStroke(tool: Tool): Stroke {
  return { tool, touched: new Set(), changes: [] };
}

// 긋는 동안 칸 하나에 도구를 적용한다. 이미 지나간 칸이거나 바뀌지 않으면 null.
export function strokeApply(board: Board, stroke: Stroke, index: number): Change | null {
  if (stroke.touched.has(index)) return null;
  stroke.touched.add(index);
  const before = board.cells[index];
  const after = applyTool(before, stroke.tool);
  if (before === after) return null;
  board.cells[index] = after;
  const change = { index, before, after };
  stroke.changes.push(change);
  return change;
}

export function bulk(board: Board, op: BulkOp): Change[] {
  const changes: Change[] = [];
  for (let index = 0; index < board.cells.length; index++) {
    const before = board.cells[index];
    const after =
      op === "flipAll"
        ? applyTool(before, "flip")
        : op === "fillAll"
          ? applyTool(before, "place")
          : EMPTY;
    if (before === after) continue;
    board.cells[index] = after;
    changes.push({ index, before, after });
  }
  return changes;
}

export function createHistory(): History {
  return { undo: [], redo: [] };
}

export function commit(history: History, changes: Change[]): void {
  if (changes.length === 0) return;
  history.undo.push(changes);
  if (history.undo.length > MAX_HISTORY) history.undo.shift();
  history.redo = [];
}

export function undo(board: Board, history: History): Change[] | null {
  const step = history.undo.pop();
  if (!step) return null;
  const applied: Change[] = [];
  for (let i = step.length - 1; i >= 0; i--) {
    const c = step[i];
    board.cells[c.index] = c.before;
    applied.push({ index: c.index, before: c.after, after: c.before });
  }
  history.redo.push(step);
  return applied;
}

export function redo(board: Board, history: History): Change[] | null {
  const step = history.redo.pop();
  if (!step) return null;
  for (const c of step) board.cells[c.index] = c.after;
  history.undo.push(step);
  return step.map((c) => ({ ...c }));
}

const HEX_COLOR = /^#[0-9a-f]{6}$/i;

// 칸 상태를 2비트씩 묶어 base64로 담는다.
export function serialize(s: SavedState): string {
  const n = s.w * s.h;
  const packed = new Uint8Array(Math.ceil(n / 4));
  for (let i = 0; i < n; i++) packed[i >> 2] |= (s.cells[i] & 3) << ((i & 3) * 2);
  let bin = "";
  for (const byte of packed) bin += String.fromCharCode(byte);
  return JSON.stringify({
    v: 1,
    w: s.w,
    h: s.h,
    front: s.front,
    back: s.back,
    cells: btoa(bin),
    muted: s.muted,
  });
}

// 형식이 하나라도 맞지 않으면 null — 호출하는 쪽은 새 판으로 시작한다.
export function deserialize(json: string | null): SavedState | null {
  if (!json) return null;
  try {
    const d = JSON.parse(json);
    if (d?.v !== 1) return null;
    if (!(SIZES as readonly number[]).includes(d.w) || d.h !== d.w) return null;
    if (typeof d.front !== "string" || !HEX_COLOR.test(d.front)) return null;
    if (typeof d.back !== "string" || !HEX_COLOR.test(d.back)) return null;
    if (typeof d.muted !== "boolean" || typeof d.cells !== "string") return null;
    const bin = atob(d.cells);
    const n = d.w * d.h;
    if (bin.length !== Math.ceil(n / 4)) return null;
    const cells = new Uint8Array(n);
    for (let i = 0; i < n; i++) {
      const v = (bin.charCodeAt(i >> 2) >> ((i & 3) * 2)) & 3;
      if (v > BACK) return null;
      cells[i] = v;
    }
    return { w: d.w, h: d.h, cells, front: d.front, back: d.back, muted: d.muted };
  } catch {
    return null;
  }
}
```

```ts
// 칩 소리. 녹음 파일 없이 짧은 노이즈 버스트와 고음 클릭을 섞어 "딱"을 만든다.
// AudioContext는 브라우저 자동 재생 정책 때문에 첫 사용자 입력(unlock) 때 만든다.
import type { Tool } from "./board";

const MIN_INTERVAL_MS = 30;

export function createChipSound() {
  let ctx: AudioContext | null = null;
  let noise: AudioBuffer | null = null;
  let muted = false;
  let last = 0;

  function ensure(): AudioContext {
    if (!ctx) {
      ctx = new AudioContext();
      noise = ctx.createBuffer(1, Math.ceil(ctx.sampleRate * 0.05), ctx.sampleRate);
      const data = noise.getChannelData(0);
      for (let i = 0; i < data.length; i++) data[i] = Math.random() * 2 - 1;
    }
    if (ctx.state === "suspended") void ctx.resume();
    return ctx;
  }

  return {
    unlock() {
      ensure();
    },
    setMuted(m: boolean) {
      muted = m;
    },
    play(kind: Tool) {
      if (muted) return;
      const nowMs = performance.now();
      if (nowMs - last < MIN_INTERVAL_MS) return;
      last = nowMs;
      const c = ensure();
      const t = c.currentTime;
      // 빼기·놓기는 조금 낮게, 칩마다 ±10% 흔든다
      const pitch = (kind === "flip" ? 1 : 0.75) * (0.9 + Math.random() * 0.2);

      const src = c.createBufferSource();
      src.buffer = noise;
      const band = c.createBiquadFilter();
      band.type = "bandpass";
      band.frequency.value = 3000 * pitch;
      band.Q.value = 1.2;
      const noiseGain = c.createGain();
      noiseGain.gain.setValueAtTime(0.5, t);
      noiseGain.gain.exponentialRampToValueAtTime(0.001, t + 0.04);
      src.connect(band).connect(noiseGain).connect(c.destination);
      src.start(t);
      src.stop(t + 0.05);

      const osc = c.createOscillator();
      osc.type = "triangle";
      osc.frequency.setValueAtTime(2400 * pitch, t);
      osc.frequency.exponentialRampToValueAtTime(900 * pitch, t + 0.02);
      const oscGain = c.createGain();
      oscGain.gain.setValueAtTime(0.25, t);
      oscGain.gain.exponentialRampToValueAtTime(0.001, t + 0.025);
      osc.connect(oscGain).connect(c.destination);
      osc.start(t);
      osc.stop(t + 0.03);
    },
    dispose() {
      void ctx?.close();
      ctx = null;
    },
  };
}
```

```ts
"use client";

import { type RefObject, useEffect, useRef } from "react";
import * as THREE from "three";
import { RoomEnvironment } from "three/examples/jsm/environments/RoomEnvironment.js";
import { type Board, type Change, BACK, EMPTY } from "./board";
import { type Point, boardSize, cellCenter } from "./layout";

const CHIP_RADIUS = 0.46;
const CHIP_THICKNESS = 0.06;
const CHIP_Y = CHIP_THICKNESS / 2;
const FOV = 35;
const TILT = THREE.MathUtils.degToRad(25);
const MAX_ZOOM = 8;
const FLIP_MS = 200;
const REMOVE_MS = 150;
const PLACE_MS = 150;
const FLIP_LIFT = 0.08;
const DROP_HEIGHT = 0.15;
const TEXTURE_PX = 24; // 바닥 텍스처: 칩 지름 1당 픽셀
const BACKGROUND = "#d9f0ff";
const BASE_COLOR = "#2b2f38";
const HOLE_COLOR = "#3c414d";

export type SceneApi = {
  setBoard(b: Board): void;
  animateChanges(changes: Change[], delayOf?: (c: Change, i: number) => number): void;
  setColors(front: string, back: string): void;
  boardPoint(clientX: number, clientY: number): Point | null;
  zoomAt(clientX: number, clientY: number, factor: number): void;
  panBy(dx: number, dy: number): void;
  fit(): void;
  exportPng(): Promise<Blob | null>;
};

type Anim = {
  index: number;
  kind: "flip" | "remove" | "place";
  fromAngle: number;
  start: number;
  dur: number;
};

const easeOut = (t: number) => 1 - (1 - t) ** 3;
const clamp = (v: number, lo: number, hi: number) => Math.min(hi, Math.max(lo, v));

// three.js 장면을 container 안에 만들고, 조작용 API를 ref로 돌려준다.
// 칩 전체가 InstancedMesh 하나이고, 애니메이션 중인 칩만 매 프레임 행렬을 고친다.
export function useChipScene(containerRef: RefObject<HTMLDivElement | null>) {
  const apiRef = useRef<SceneApi | null>(null);

  useEffect(() => {
    const container = containerRef.current;
    if (!container) return;

    const renderer = new THREE.WebGLRenderer({ antialias: true });
    renderer.setPixelRatio(Math.min(window.devicePixelRatio, 2));
    renderer.outputColorSpace = THREE.SRGBColorSpace;
    renderer.domElement.style.display = "block";
    renderer.domElement.style.width = "100%";
    renderer.domElement.style.height = "100%";
    container.appendChild(renderer.domElement);

    const scene = new THREE.Scene();
    scene.background = new THREE.Color(BACKGROUND);
    const pmrem = new THREE.PMREMGenerator(renderer);
    const envTexture = pmrem.fromScene(new RoomEnvironment(), 0.04).texture;
    scene.environment = envTexture;
    const light = new THREE.DirectionalLight(0xffffff, 1.2);
    light.position.set(-3, 8, 4);
    scene.add(light);

    const camera = new THREE.PerspectiveCamera(FOV, 1, 0.1, 500);
    const target = new THREE.Vector3();
    let fitDistance = 10;
    let zoom = 1;

    const geometry = new THREE.CylinderGeometry(CHIP_RADIUS, CHIP_RADIUS, CHIP_THICKNESS, 32);
    const chipMaterial = () => new THREE.MeshStandardMaterial({ metalness: 0.6, roughness: 0.35 });
    // CylinderGeometry 그룹 순서: 옆면, 윗면(앞면), 아랫면(뒷면)
    const sideMat = chipMaterial();
    const frontMat = chipMaterial();
    const backMat = chipMaterial();
    const baseMat = new THREE.MeshStandardMaterial({ roughness: 0.9, metalness: 0 });
    const baseMesh = new THREE.Mesh(new THREE.PlaneGeometry(1, 1), baseMat);
    baseMesh.rotation.x = -Math.PI / 2;
    scene.add(baseMesh);
    let baseTexture: THREE.CanvasTexture | null = null;
    let chips: THREE.InstancedMesh | null = null;
    let board: Board | null = null;
    const anims = new Map<number, Anim>();
    let needsRender = true;
    const reduceMotion = window.matchMedia("(prefers-reduced-motion: reduce)").matches;

    const matrix = new THREE.Matrix4();
    const quat = new THREE.Quaternion();
    const euler = new THREE.Euler();
    const pos = new THREE.Vector3();
    const scale = new THREE.Vector3();

    function setChipMatrix(i: number, angle: number, lift: number, s: number) {
      const b = board!;
      const c = cellCenter(i % b.w, Math.floor(i / b.w), b.w, b.h);
      pos.set(c.x, CHIP_Y + lift, c.y);
      euler.set(angle, 0, 0);
      quat.setFromEuler(euler);
      scale.setScalar(s);
      matrix.compose(pos, quat, scale);
      chips!.setMatrixAt(i, matrix);
    }

    function restMatrix(i: number) {
      const s = board!.cells[i];
      setChipMatrix(i, s === BACK ? Math.PI : 0, 0, s === EMPTY ? 0 : 1);
    }

    // 판 바닥과 칩이 없는 자리의 홈을 캔버스 텍스처 한 장에 그린다.
    function buildBase(b: Board) {
      const { width, height } = boardSize(b.w, b.h);
      const W = width + 1;
      const H = height + 1;
      const canvas = document.createElement("canvas");
      canvas.width = Math.ceil(W * TEXTURE_PX);
      canvas.height = Math.ceil(H * TEXTURE_PX);
      const ctx = canvas.getContext("2d")!;
      ctx.fillStyle = BASE_COLOR;
      ctx.fillRect(0, 0, canvas.width, canvas.height);
      ctx.fillStyle = HOLE_COLOR;
      for (let row = 0; row < b.h; row++) {
        for (let col = 0; col < b.w; col++) {
          const c = cellCenter(col, row, b.w, b.h);
          ctx.beginPath();
          ctx.arc((c.x + W / 2) * TEXTURE_PX, (c.y + H / 2) * TEXTURE_PX, 0.42 * TEXTURE_PX, 0, Math.PI * 2);
          ctx.fill();
        }
      }
      baseTexture?.dispose();
      baseTexture = new THREE.CanvasTexture(canvas);
      baseTexture.colorSpace = THREE.SRGBColorSpace;
      baseTexture.anisotropy = renderer.capabilities.getMaxAnisotropy();
      baseMat.map = baseTexture;
      baseMat.needsUpdate = true;
      baseMesh.scale.set(W, H, 1);
    }

    function updateCamera() {
      const dist = fitDistance / zoom;
      camera.position.set(target.x, dist * Math.cos(TILT), target.z + dist * Math.sin(TILT));
      camera.lookAt(target);
      camera.updateMatrixWorld();
      needsRender = true;
    }

    function fit() {
      if (!board) return;
      const { width, height } = boardSize(board.w, board.h);
      const half = Math.tan(THREE.MathUtils.degToRad(FOV) / 2);
      fitDistance = Math.max((height + 1) / (2 * half), (width + 1) / (2 * half * camera.aspect)) * 1.05;
      zoom = 1;
      target.set(0, 0, 0);
      updateCamera();
    }

    function clampTarget() {
      if (!board) return;
      const { width, height } = boardSize(board.w, board.h);
      target.x = clamp(target.x, -width / 2, width / 2);
      target.z = clamp(target.z, -height / 2, height / 2);
    }

    const raycaster = new THREE.Raycaster();
    const chipPlane = new THREE.Plane(new THREE.Vector3(0, 1, 0), -CHIP_Y);
    const hit = new THREE.Vector3();
    const ndc = new THREE.Vector2();

    // 화면 좌표 → 칩 높이 평면 위의 점(layout 좌표). 지평선 위라 만나지 않으면 null.
    function boardPoint(clientX: number, clientY: number): Point | null {
      const r = renderer.domElement.getBoundingClientRect();
      if (!r.width || !r.height) return null;
      ndc.set(((clientX - r.left) / r.width) * 2 - 1, -((clientY - r.top) / r.height) * 2 + 1);
      raycaster.setFromCamera(ndc, camera);
      return raycaster.ray.intersectPlane(chipPlane, hit) ? { x: hit.x, y: hit.z } : null;
    }

    // 포인터 아래 지점이 그대로 머물도록 확대한다.
    function zoomAt(clientX: number, clientY: number, factor: number) {
      const before = boardPoint(clientX, clientY);
      zoom = clamp(zoom * factor, 1, MAX_ZOOM);
      updateCamera();
      const after = boardPoint(clientX, clientY);
      if (before && after) {
        target.x += before.x - after.x;
        target.z += before.y - after.y;
      }
      clampTarget();
      updateCamera();
    }

    function panBy(dx: number, dy: number) {
      const h = renderer.domElement.clientHeight || 1;
      const dist = fitDistance / zoom;
      const worldPerPx = (2 * dist * Math.tan(THREE.MathUtils.degToRad(FOV) / 2)) / h;
      target.x -= dx * worldPerPx;
      target.z -= (dy * worldPerPx) / Math.cos(TILT);
      clampTarget();
      updateCamera();
    }

    function setBoard(b: Board) {
      board = b;
      anims.clear();
      const n = b.w * b.h;
      if (!chips || chips.count !== n) {
        if (chips) {
          scene.remove(chips);
          chips.dispose();
        }
        chips = new THREE.InstancedMesh(geometry, [sideMat, frontMat, backMat], n);
        chips.instanceMatrix.setUsage(THREE.DynamicDrawUsage);
        // 인스턴스 하나의 경계로 판 전체를 잘라 내지 않도록 끈다
        chips.frustumCulled = false;
        scene.add(chips);
      }
      for (let i = 0; i < n; i++) restMatrix(i);
      chips.instanceMatrix.needsUpdate = true;
      buildBase(b);
      needsRender = true;
    }

    function animateChanges(changes: Change[], delayOf: (c: Change, i: number) => number = () => 0) {
      if (!board || !chips) return;
      const now = performance.now();
      changes.forEach((c, i) => {
        if (reduceMotion) {
          anims.delete(c.index);
          restMatrix(c.index);
          return;
        }
        const kind = c.after === EMPTY ? "remove" : c.before === EMPTY ? "place" : "flip";
        const fromAngle = c.before === BACK ? Math.PI : 0;
        const dur = kind === "flip" ? FLIP_MS : kind === "remove" ? REMOVE_MS : PLACE_MS;
        anims.set(c.index, { index: c.index, kind, fromAngle, start: now + delayOf(c, i), dur });
        // 차례가 올 때까지는 바뀌기 전 모습으로 둔다
        setChipMatrix(c.index, fromAngle, 0, c.before === EMPTY ? 0 : 1);
      });
      chips.instanceMatrix.needsUpdate = true;
      needsRender = true;
    }

    function setColors(front: string, back: string) {
      frontMat.color.set(front);
      backMat.color.set(back);
      sideMat.color.set(front).lerp(new THREE.Color(back), 0.5).multiplyScalar(0.55);
      needsRender = true;
    }

    function exportPng(): Promise<Blob | null> {
      renderer.render(scene, camera);
      return new Promise((resolve) => renderer.domElement.toBlob(resolve, "image/png"));
    }

    let raf = 0;
    const tick = () => {
      raf = requestAnimationFrame(tick);
      if (anims.size > 0 && chips && board) {
        const now = performance.now();
        for (const a of anims.values()) {
          const t = (now - a.start) / a.dur;
          if (t < 0) continue;
          if (t >= 1) {
            restMatrix(a.index);
            anims.delete(a.index);
            continue;
          }
          const k = easeOut(t);
          if (a.kind === "flip")
            setChipMatrix(a.index, a.fromAngle + Math.PI * k, Math.sin(Math.PI * t) * FLIP_LIFT, 1);
          else if (a.kind === "remove") setChipMatrix(a.index, a.fromAngle, k * DROP_HEIGHT, 1 - k);
          else setChipMatrix(a.index, 0, (1 - k) * DROP_HEIGHT, k);
        }
        chips.instanceMatrix.needsUpdate = true;
        needsRender = true;
      }
      if (needsRender) {
        renderer.render(scene, camera);
        needsRender = false;
      }
    };
    raf = requestAnimationFrame(tick);

    const resize = () => {
      const w = container.clientWidth;
      const h = container.clientHeight;
      if (!w || !h) return;
      renderer.setSize(w, h, false);
      camera.aspect = w / h;
      camera.updateProjectionMatrix();
      // 사용자가 확대·이동하지 않았다면 새 크기에 맞춰 다시 맞춘다
      if (zoom === 1 && target.lengthSq() === 0) fit();
      else updateCamera();
    };
    const ro = new ResizeObserver(resize);
    ro.observe(container);
    resize();

    apiRef.current = { setBoard, animateChanges, setColors, boardPoint, zoomAt, panBy, fit, exportPng };

    return () => {
      apiRef.current = null;
      cancelAnimationFrame(raf);
      ro.disconnect();
      chips?.dispose();
      geometry.dispose();
      for (const m of [sideMat, frontMat, backMat, baseMat]) m.dispose();
      baseMesh.geometry.dispose();
      baseTexture?.dispose();
      envTexture.dispose();
      pmrem.dispose();
      renderer.dispose();
      renderer.domElement.remove();
    };
  }, [containerRef]);

  return apiRef;
}
```

```tsx
  CirclePlus,

  { op: "clearAll", label: "모두 빼기" },

  const d = new Date();
  const p = (n: number) => String(n).padStart(2, "0");
  return `${d.getFullYear()}${p(d.getMonth() + 1)}${p(d.getDate())}-${p(d.getHours())}${p(d.getMinutes())}`;

export default function Monorism() {
  const containerRef = useRef<HTMLDivElement>(null);
  const scene = useChipScene(containerRef);
  const [sound] = useState(createChipSound);
  const boardRef = useRef(createBoard(DEFAULT_SIZE, DEFAULT_SIZE));
  const historyRef = useRef(createHistory());

  const lastPointRef = useRef<Point | null>(null);
  const panRef = useRef<{ x: number; y: number } | null>(null);
  const touchesRef = useRef(new Map<number, { x: number; y: number }>());
  const saveTimer = useRef<ReturnType<typeof setTimeout> | null>(null);

  const [tool, setTool] = useState<Tool>("flip");
  const [front, setFront] = useState(PALETTES[0].front);
  const [back, setBack] = useState(PALETTES[0].back);
  const [muted, setMuted] = useState(false);
  const [newSize, setNewSize] = useState<number>(DEFAULT_SIZE);
  const [boardVersion, setBoardVersion] = useState(0);
  const [hydrated, setHydrated] = useState(false);
  const [historyState, setHistoryState] = useState({ canUndo: false, canRedo: false });

  // 저장된 판을 한 번 불러온다. 서버 렌더와 첫 화면을 맞추려고 effect에서 읽는다.
  useEffect(() => {
    let saved = null;
    try {
      saved = deserialize(localStorage.getItem(STORAGE_KEY));
    } catch {
      saved = null;
    }
    /* eslint-disable react-hooks/set-state-in-effect */
    if (saved) {
      boardRef.current = { w: saved.w, h: saved.h, cells: saved.cells };
      setFront(saved.front);
      setBack(saved.back);
      setMuted(saved.muted);
      setNewSize(saved.w);
    }
    setHydrated(true);
    /* eslint-enable react-hooks/set-state-in-effect */
  }, []);

  useEffect(() => {
    if (!hydrated) return;
    scene.current?.setBoard(boardRef.current);
    scene.current?.fit();
  }, [hydrated, boardVersion, scene]);

  useEffect(() => {
    scene.current?.setColors(front, back);
  }, [front, back, scene]);

  useEffect(() => {
    sound.setMuted(muted);
  }, [muted, sound]);

  useEffect(() => () => sound.dispose(), [sound]);

  const scheduleSave = useCallback(() => {
    if (saveTimer.current) clearTimeout(saveTimer.current);
    saveTimer.current = setTimeout(() => {
      const b = boardRef.current;
      try {
        localStorage.setItem(
          STORAGE_KEY,
          serialize({ w: b.w, h: b.h, cells: b.cells, front, back, muted }),
        );
      } catch {
        // 저장 공간이 없거나 막혀 있으면 조용히 넘어간다
      }
    }, SAVE_DELAY_MS);
  }, [front, back, muted]);

  useEffect(() => {
    if (hydrated) scheduleSave();
  }, [hydrated, scheduleSave]);

  useEffect(
    () => () => {
      if (saveTimer.current) clearTimeout(saveTimer.current);
    },
    [],
  );

  const refreshHistory = useCallback(() => {
    const h = historyRef.current;
    setHistoryState({ canUndo: h.undo.length > 0, canRedo: h.redo.length > 0 });
  }, []);

  const showChanges = useCallback(
    (changes: Change[], kind: Tool, delayOf?: (c: Change, i: number) => number) => {
      if (changes.length === 0) return;
      scene.current?.animateChanges(changes, delayOf);
      sound.play(kind);
    },
    [scene, sound],
  );

  const doUndo = useCallback(() => {
    const changes = undo(boardRef.current, historyRef.current);
    if (!changes) return;
    showChanges(changes, "flip");
    refreshHistory();
    scheduleSave();
  }, [showChanges, refreshHistory, scheduleSave]);

  const doRedo = useCallback(() => {
    const changes = redo(boardRef.current, historyRef.current);
    if (!changes) return;
    showChanges(changes, "flip");
    refreshHistory();
    scheduleSave();
  }, [showChanges, refreshHistory, scheduleSave]);

  function strokeTo(p: Point | null) {
    const stroke = strokeRef.current;
    if (!stroke) return;
    // 포인터가 판 평면을 벗어나면 경로를 끊는다 — 돌아왔을 때 사이를 잇지 않는다
    if (!p) {
      lastPointRef.current = null;
      return;
    }
    const b = boardRef.current;
    const points = lastPointRef.current ? interpolate(lastPointRef.current, p) : [p];
    const changed: Change[] = [];
    for (const q of points) {
      const cell = cellAt(q, b.w, b.h);
      if (!cell) continue;
      const change = strokeApply(b, stroke, cell.row * b.w + cell.col);
      if (change) changed.push(change);
    }
    lastPointRef.current = p;
    showChanges(changed, stroke.tool, (_, i) => i * STROKE_STAGGER_MS);
  }

  function endStroke() {
    const stroke = strokeRef.current;
    strokeRef.current = null;
    lastPointRef.current = null;
    if (!stroke || stroke.changes.length === 0) return;
    commit(historyRef.current, stroke.changes);
    refreshHistory();
    scheduleSave();
  }

  function runBulk(op: BulkOp) {
    const b = boardRef.current;
    const changes = bulk(b, op);
    if (changes.length === 0) return;
    commit(historyRef.current, changes);
    sound.unlock();
    showChanges(changes, op === "clearAll" ? "remove" : op === "fillAll" ? "place" : "flip", (c) =>
      Math.floor(c.index / b.w) * BULK_ROW_STAGGER_MS,
    );
    refreshHistory();
    scheduleSave();
  }

  function newBoard() {
    if (!window.confirm("지금 판을 지우고 새 판을 만들까요?")) return;
    boardRef.current = createBoard(newSize, newSize);
    historyRef.current = createHistory();
    refreshHistory();
    setBoardVersion((v) => v + 1);
    scheduleSave();
  }

  async function savePng() {
    const blob = await scene.current?.exportPng();
    if (!blob) return;
    const url = URL.createObjectURL(blob);
    const a = document.createElement("a");
    a.href = url;
    a.download = `monorism-${timestamp()}.png`;
    a.click();

  useEffect(() => {

      e.preventDefault();

      if (t && (t.tagName === "INPUT" || t.tagName === "SELECT" || t.tagName === "TEXTAREA")) return;

  }, [doUndo, doRedo]);

  function onPointerDown(e: React.PointerEvent<HTMLDivElement>) {
    if (e.pointerType === "touch") {
      touchesRef.current.set(e.pointerId, { x: e.clientX, y: e.clientY });

      if (touchesRef.current.size >= 2) {

    sound.unlock();
    strokeRef.current = beginStroke(tool);
    lastPointRef.current = null;
    strokeTo(scene.current?.boardPoint(e.clientX, e.clientY) ?? null);
  }

  function onPointerMove(e: React.PointerEvent<HTMLDivElement>) {
    const touches = touchesRef.current;
    if (e.pointerType === "touch" && touches.has(e.pointerId)) {
      if (touches.size >= 2) {
        const [a0, b0] = [...touches.values()];
        touches.set(e.pointerId, { x: e.clientX, y: e.clientY });
        const [a1, b1] = [...touches.values()];
        const d0 = Math.hypot(a0.x - b0.x, a0.y - b0.y);
        const d1 = Math.hypot(a1.x - b1.x, a1.y - b1.y);
        const mx0 = (a0.x + b0.x) / 2;
        const my0 = (a0.y + b0.y) / 2;
        const mx1 = (a1.x + b1.x) / 2;
        const my1 = (a1.y + b1.y) / 2;
        scene.current?.panBy(mx1 - mx0, my1 - my0);

        return;
      }
      touches.set(e.pointerId, { x: e.clientX, y: e.clientY });

      scene.current?.panBy(e.clientX - panRef.current.x, e.clientY - panRef.current.y);

      return;
    }
    if (strokeRef.current) strokeTo(scene.current?.boardPoint(e.clientX, e.clientY) ?? null);
  }

    panRef.current = null;

          ))}

            onChange={(e) => setFront(e.target.value)}
            className="h-6 w-7 cursor-pointer rounded border-2 border-point"

            aria-label="뒷면 색"
            value={back}
            onChange={(e) => setBack(e.target.value)}
            className="h-6 w-7 cursor-pointer rounded border-2 border-point"

            value={newSize}
            onChange={(e) => setNewSize(Number(e.target.value))}
            className="rounded-md border-2 border-point bg-white px-1 py-0.5 text-xs font-bold text-point"
          >
            {SIZES.map((s) => (

                {s}×{s}
              </option>
            ))}
          </select>
          <button type="button" onClick={newBoard} className={BUTTON}>
            새 판

        </div>

          <button

            title="화면 맞춤"
            aria-label="화면 맞춤"
            onClick={() => scene.current?.fit()}
            className={BUTTON}
          >
            <Maximize aria-hidden className="size-4" />
          </button>
          <button
            type="button"
            title={muted ? "소리 켜기" : "소리 끄기"}
            aria-label={muted ? "소리 켜기" : "소리 끄기"}
            onClick={() => setMuted((m) => !m)}
            className={BUTTON}
          >
            {muted ? <VolumeX aria-hidden className="size-4" /> : <Volume2 aria-hidden className="size-4" />}
          </button>
          <button type="button" onClick={savePng} className={BUTTON}>
            <Download aria-hidden className="size-4" />
            PNG로 저장
          </button>
        </div>
      </div>

      <div
        ref={containerRef}
        aria-label="모노리즘 판"
        role="application"
        className="relative min-h-0 flex-1 touch-none select-none"
        onPointerDown={onPointerDown}
        onPointerMove={onPointerMove}
        onPointerUp={onPointerEnd}
        onPointerCancel={onPointerEnd}
        onContextMenu={(e) => e.preventDefault()}
      />
    </div>
  );
}
```

- [ ] **Step 3: Playwright 검증**

dev 서버(`npm run dev:services`)가 떠 있는 상태에서 세션 스크래치 디렉터리에 아래 파일을 만들어 실행한다. `REPO`는 레포 절대 경로로 바꾼다.

```js
import { chromium } from "REPO/node_modules/playwright/index.mjs";
import assert from "node:assert/strict";

const URL = "http://localhost:3100/monorism";
const b = await chromium.launch();
const ctx = await b.newContext({ viewport: { width: 1280, height: 900 }, acceptDownloads: true });
const p = await ctx.newPage();
const errors = [];
p.on("console", (m) => m.type() === "error" && errors.push(m.text()));
p.on("pageerror", (e) => errors.push(e.message));

// 저장값을 읽어 상태별 칸 수를 센다
async function counts() {
  await p.waitForTimeout(700);
  const json = await p.evaluate(() => localStorage.getItem("monorism-board"));
  const d = JSON.parse(json);
  const bin = Buffer.from(d.cells, "base64");
  const c = [0, 0, 0];
  for (let i = 0; i < d.w * d.h; i++) c[(bin[i >> 2] >> ((i & 3) * 2)) & 3]++;
  return { w: d.w, empty: c[0], front: c[1], back: c[2] };
}

await p.goto(URL, { waitUntil: "networkidle" });
await p.evaluate(() => localStorage.clear());
await p.reload({ waitUntil: "networkidle" });
const boardEl = p.getByLabel("모노리즘 판");
const box = await boardEl.boundingBox();
const cx = box.x + box.width / 2;
const cy = box.y + box.height / 2;

// 클릭 한 번 → 뒷면 1개
await p.mouse.click(cx, cy);
let s = await counts();
assert.equal(s.w, 40);
assert.equal(s.back, 1);

// 가로로 긋기 → 여러 개, 같은 칸은 한 번만(되돌아와도 늘지 않음)
await p.mouse.move(cx - 150, cy + 60);
await p.mouse.down();
await p.mouse.move(cx + 150, cy + 60, { steps: 12 });
await p.mouse.move(cx - 150, cy + 60, { steps: 12 });
await p.mouse.up();
s = await counts();
assert.ok(s.back > 5, `back ${s.back}`);
const afterStroke = s.back;

// 실행 취소 → 클릭 한 번 상태로
await p.keyboard.press("Control+z");
s = await counts();
assert.equal(s.back, 1);
await p.keyboard.press("Control+Shift+z");
s = await counts();
assert.equal(s.back, afterStroke);
await p.keyboard.press("Control+z");

// 빼기·놓기
await p.keyboard.press("2");
await p.mouse.click(cx, cy - 120);
s = await counts();
assert.equal(s.empty, 1);
await p.keyboard.press("3");
await p.mouse.click(cx, cy - 120);
s = await counts();
assert.equal(s.empty, 0);

// 새로고침 후 유지
await p.reload({ waitUntil: "networkidle" });
s = await counts();
assert.equal(s.back, 1);

// 새 판 24×24
await p.getByLabel("새 판 크기").selectOption("24");
p.once("dialog", (d) => d.accept());
await p.getByRole("button", { name: "새 판" }).click();
s = await counts();
assert.equal(s.w, 24);
assert.equal(s.back, 0);

// 모두 빼기 → 전부 빈 자리
await p.getByRole("button", { name: "모두 빼기" }).click();
s = await counts();
assert.equal(s.empty, 24 * 24);

// PNG 저장
const [download] = await Promise.all([
  p.waitForEvent("download"),
  p.getByRole("button", { name: "PNG로 저장" }).click(),
]);
assert.match(download.suggestedFilename(), /^monorism-\d{8}-\d{4}\.png$/);

// 모바일 폭 가로 스크롤 없음
const m = await ctx.newPage();
await m.setViewportSize({ width: 375, height: 812 });
await m.goto(URL, { waitUntil: "networkidle" });
assert.equal(await m.evaluate(() => document.documentElement.scrollWidth > innerWidth), false);

await b.close();
assert.deepEqual(errors, [], errors.join("\n"));
console.log("OK");
```
